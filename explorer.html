d3 = require("d3-random");
maxTime = 72;
nPoints = Math.floor(maxTime / 0.25);
S1Range = [0, 20];
S2Range = [-15, 5];
a1Range = [0.1, 10];
a2Range = [0, 10];
b1Range = [5, maxTime];
b2Range = [-20, maxTime];

function getRandom(min, max) {
  return Math.random() * (max - min + 1) + min;
}

function generateRawData() {
  const S1 = +getRandom(10, 20).toFixed(2);
  const S2 = +getRandom(-5, 0).toFixed(2);
  const a1 = +getRandom(0.15, 10).toFixed(2);
  const a2 = +getRandom(0, 0.7).toFixed(2);
  const b1 = +getRandom(10, 60).toFixed(2);
  const b2 = +getRandom(0.9 * b1, 1.1 * b1).toFixed(2);
  const points = Array.from({length: nPoints}, (_, i) => {
    const t = i * maxTime / (nPoints - 1);
    return (S1 / (1 + Math.exp(a1 * (b1 - t)))) +
           (S2 / (1 + Math.exp(a2 * (b2 - t)))) +
           d3.randomNormal(0, 0.1)();
  });
  return {params: {S1, S2, a1, a2, b1, b2}, points};
}

mutable rawData = generateRawData()
mutable showAnswer = false

curve = Array.from({length: nPoints}, (_, i) => {
  const t = i * maxTime / (nPoints - 1);
  const g1 = S1 / (1 + Math.exp(a1 * (b1 - t)));
  const g2 = S2 / (1 + Math.exp(a2 * (b2 - t)));
  const f = g1 + g2;
  const raw = rawData.points[i];
  return {t, g1, g2, f, raw, resid: f - raw};
})
